pub fn fold(cp: u21, output: *[3]u21) u2 {
    if (cp < 128) {
        output[0] = if (cp >= 'A' and cp <= 'Z') cp + 32 else cp;
        return 1;
    }
    var lo: usize = 0;
    var hi: usize = ranges.len;
    while (lo < hi) {
        const mid = lo + (hi - lo) / 2;
        const range = ranges[mid];
        if (cp < range.first) {
            hi = mid;
        } else if (cp > range.last) {
            lo = mid + 1;
        } else {
            if ((cp - range.first) % range.stride == 0) {
                output[0] = @intCast(@as(i32, cp) + range.delta);
                return 1;
            }
            break;
        }
    }
    lo = 0;
    hi = expansions.len;
    while (lo < hi) {
        const mid = lo + (hi - lo) / 2;
        const item = expansions[mid];
        if (cp < item.key) {
            hi = mid;
        } else if (cp > item.key) {
            lo = mid + 1;
        } else {
            output.* = item.values;
            return item.len;
        }
    }
    output[0] = cp;
    return 1;
}

test "compressed lookup agrees on every Unicode code point, including holes" {
    const std = @import("std");
    var r: usize = 0;
    var e: usize = 0;
    var cp: u21 = 0;
    while (cp <= 0x10ffff) : (cp += 1) {
        while (r < ranges.len and cp > ranges[r].last) : (r += 1) {}
        while (e < expansions.len and cp > expansions[e].key) : (e += 1) {}
        var expected = [3]u21{ cp, 0, 0 };
        var len: u2 = 1;
        if (r < ranges.len and cp >= ranges[r].first and (cp - ranges[r].first) % ranges[r].stride == 0) {
            expected[0] = @intCast(@as(i32, cp) + ranges[r].delta);
        } else if (e < expansions.len and cp == expansions[e].key) {
            expected = expansions[e].values;
            len = expansions[e].len;
        }
        var actual: [3]u21 = undefined;
        try std.testing.expectEqual(len, fold(cp, &actual));
        try std.testing.expectEqualSlices(u21, expected[0..len], actual[0..len]);
    }
}
